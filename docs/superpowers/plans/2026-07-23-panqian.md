```python
# -*- coding: utf-8 -*-
"""共享请求 helper + 结果容器。

请求模式对齐 fupan/main.py:屏蔽环境代理 + trust_env=False(国内 API 直连)。
Yahoo fallback 单独走代理(见 proxy_get)。
"""
import os
import re
import requests

# 屏蔽环境代理(与 fupan/market_regime 一致,避免本地代理干扰国内 API 直连)
for _k in ("HTTP_PROXY", "HTTPS_PROXY", "http_proxy", "https_proxy"):
    os.environ.pop(_k, None)
os.environ["NO_PROXY"] = "*"

# 直连 session(新浪/财联社/金十/新华)
sess = requests.Session()
sess.trust_env = False
sess.headers.update({"Referer": "https://finance.sina.com.cn"})   # 新浪 hq.sinajs.cn 需 Referer

SINA_HQ = "https://hq.sinajs.cn/list="


class FetchResult:
    """单维采集结果。ok=False 时 detail 写失败原因,供报告取数质量栏使用。"""
    def __init__(self, dim, ok=False, data=None, detail=""):
        self.dim = dim        # "us" / "a50" / "fx" / "news" ...
        self.ok = ok
        self.data = data      # 维度自定义结构
        self.detail = detail  # 成功="✓";失败=原因


def sina_quote(codes):
    """拉新浪 hq.sinajs.cn 行情。返回 dict: code -> 原始字段 list(逗号分隔)。
    例: {'gb_ixic': ['纳斯达克','18000.00', ...]}
    解析失败/网络异常返回 {}(不抛)。
    code 无效时新浪返回 'var gb_xxx="";'(空串),调用方据此判空。"""
    try:
        r = sess.get(SINA_HQ + ",".join(codes), timeout=15)
        r.encoding = "gbk"                       # 新浪行情是 gbk
        out = {}
        for line in r.text.splitlines():
            m = re.match(r'var\s+(\w+)="([^"]*)"', line.strip())
            if not m:
                continue
            code, payload = m.group(1), m.group(2)
            out[code] = payload.split(",") if payload else []
        return out
    except Exception:
        return {}


def proxy_get(url, proxy, **kw):
    """Yahoo 等 fallback 用:显式走代理。超时/异常返回 None。"""
    try:
        r = requests.get(url, proxies=proxy, timeout=20, **kw)
        r.raise_for_status()
        return r
    except Exception:
        return None
```

```python
# -*- coding: utf-8 -*-
"""维度1:美股三大指数 + VIX。新浪为主,Yahoo fallback。"""
import re
from pk import base
from pk.config import US_INDICES, US_VIX, PROXY

# 字段索引 — 以 probes/probe_all.py us 的输出为准校准(Step 2)
IDX_NAME = 0
IDX_PRICE = 1
IDX_PCT = 3          # 涨跌幅%; probe 后校准
NAME_MAP = dict(US_INDICES + [US_VIX])   # code -> 展示名


def parse_us(raw_text):
    """从新浪 hq 返回文本解析出 [{code,name,price,pct}]。空 payload 跳过。"""
    out = []
    for m in re.finditer(r'var\s+(\w+)="([^"]*)"', raw_text):
        code, payload = m.group(1), m.group(2)
        if not payload or code not in NAME_MAP:
            continue
        f = payload.split(",")
        try:
            price = float(f[IDX_PRICE])
            pct = float(f[IDX_PCT])
        except (IndexError, ValueError):
            continue
        out.append({"code": code, "name": NAME_MAP[code], "price": price, "pct": pct})
    return out


def _yahoo_pct(symbol):
    """Yahoo fallback:取涨跌幅(走代理)。返回 float 或 None。"""
    r = base.proxy_get(f"https://query1.finance.yahoo.com/v8/finance/chart/{symbol}",
                       proxy=PROXY)
    if not r:
        return None
    try:
        j = r.json()
        meta = j["chart"]["result"][0]["meta"]
        return meta.get("regularMarketChangePercent")
    except Exception:
        return None


def fetch_us():
    """返回 FetchResult(dim='us')。data=[{name,price,pct}], VIX 单列 data['vix']。"""
    codes = [c for c, _ in US_INDICES]
    raw = base.sina_quote(codes)
    # 拼回文本走 parse_us(复用解析)
    text = "".join(f'var {c}="{",".join(raw.get(c, []))}";' for c in codes)
    items = parse_us(text)
    # Yahoo 补缺失指数
    have = {it["code"] for it in items}
    ym = {"gb_ixic": "^IXIC", "gb_inx": "^GSPC", "gb_dji": "^DJI"}
    for c, _ in US_INDICES:
        if c not in have:
            p = _yahoo_pct(ym.get(c, ""))
            if p is not None:
                items.append({"code": c, "name": NAME_MAP[c], "price": 0.0, "pct": float(p)})
    # VIX(新浪无则 Yahoo)
    vraw = base.sina_quote([US_VIX[0]])
    vp = vraw.get(US_VIX[0])
    vix = None
    if vp and len(vp) > IDX_PCT:
        try:
            vix = float(vp[IDX_PRICE])
        except ValueError:
            vix = None
    if vix is None:
        p = _yahoo_pct("^VIX")
        vix = p if p is not None else None
    ok = bool(items)
    return base.FetchResult("us", ok=ok, data={"indices": items, "vix": vix},
                            detail="✓" if ok else "美股取数失败(新浪+Yahoo 均失败)")
```

```python
from pk import base

# 期货 hf_ 字段索引 — probe 校准
IDX_PRICE_HF = 0     # hf_ 期货首字段常为最新价,probe 后校准

IDX_PRICE_GB = 1     # 中概 gb_ 复用美股式(probe 校准)

def _pick(fields, idx_price, idx_pct):
    try:
        return {"price": float(fields[idx_price]), "pct": float(fields[idx_pct])}
    except (IndexError, ValueError):
        return None


def parse_a50_cnr(raw_text):
    """返回 {'a50': {price,pct}|None, 'cnr': [{name,pct}]}。"""
    fields_map = {}
    for m in re.finditer(r'var\s+(\w+)="([^"]*)"', raw_text):
        code, payload = m.group(1), m.group(2)
        fields_map[code] = payload.split(",") if payload else []

    af = fields_map.get(A50_CODE[0], [])
    if af:
        a50 = _pick(af, IDX_PRICE_HF, IDX_PCT_HF)

    cnr = []

    for code, name in [CNR_DRAGON] + CNR_STOCKS:
        f = fields_map.get(code, [])
        if not f:
            continue
        p = _pick(f, IDX_PRICE_GB, IDX_PCT_GB)
        if p:
            cnr.append({"name": name, "pct": p["pct"]})
    return {"a50": a50, "cnr": cnr}
```

```python
"""维度3:离岸人民币 + 大宗(金/油/铜)。板块映射由 render 解读。"""
import re
from pk import base
from pk.config import FX, COMMODITY

IDX_PCT = 3   # probe 校准


def _parse_group(raw_text, pairs, idx_pct):
    out = []
    fmap = {}
    for m in re.finditer(r'var\s+(\w+)="([^"]*)"', raw_text):
        fmap[m.group(1)] = m.group(2).split(",") if m.group(2) else []

        if len(f) > idx_pct:
            try:
                out.append({"name": name, "pct": float(f[idx_pct])})
            except ValueError:
                pass
    return out


def parse_fx_comm(raw_text):

    d = parse_fx_comm(text)
    ok = bool(d["fx"] or d["comm"])
```

```python
"""维度4:国际重大事件。财联社+金十+新华财经,时间窗(昨夜18:00~今晨)+关键词过滤+去重。
自动抓不可能 100% 可靠,报告底部留手动补充兜口(由 render 体现)。"""
import time
from datetime import datetime, timedelta
from difflib import SequenceMatcher
from pk import base
from pk.config import NEWS_KEYWORDS, POLICY_KEYWORDS

CLS_URL = ("https://www.cls.cn/nodeapi/updateTelegraphList?"
           "app=CailianpressWeb&category=&lastTime=&os=web&sv=7.7.5&rn=80")
CLS_HEADERS = {"Referer": "https://www.cls.cn/telegraph"}


def _window_start_ts(now=None):
    """时间窗起点=昨日18:00(北京)。返回秒级时间戳。"""
    now = now or datetime.now()
    start = (now - timedelta(days=1)).replace(hour=18, minute=0, second=0, microsecond=0)
    return start.timestamp()


def _hit_keyword(text):
    kw = NEWS_KEYWORDS + POLICY_KEYWORDS
    t = text or ""
    return any(k in t for k in kw)


def parse_cls(data, now=None):
    """财联社:取 roll_data,过滤时间窗+关键词。返回 [{title,ts,source,star}]。"""
    items = (data.get("data", {}) or {}).get("roll_data", []) or data.get("data", [])
    cutoff = _window_start_ts(now)
    out = []
    for it in items:
        ts = it.get("ctime") or it.get("time")
        if not ts or ts < cutoff:
            continue
        title = it.get("title") or ""
        content = it.get("content") or ""
        if not _hit_keyword(title + content):
            continue
        star = 3 if it.get("level") in (2, "2", "B") else (2 if it.get("level") in (1, "1") else 1)
        out.append({"title": title.strip()[:80], "ts": ts, "source": "财联社", "star": star})
    return out


def parse_jin10(data, now=None):
    """金十财经日历:取重要事件(星级>=2)。字段名以 fixture 为准校准。"""
    cutoff = _window_start_ts(now)
    out = []
    for it in data.get("items", []) or data.get("data", []) or []:
        ts = it.get("time") or it.get("date")
        star = it.get("star") or it.get("importance") or 0
        try:
            star = int(star)
        except (TypeError, ValueError):
            star = 0
        title = it.get("title") or it.get("event") or ""
        if star < 2 or not _hit_keyword(title):
            continue
        if ts and isinstance(ts, (int, float)) and ts < cutoff:
            continue
        out.append({"title": title.strip()[:80], "ts": ts or 0, "source": "金十", "star": star})
    return out


def parse_xinhua(data, now=None):
    """新华财经:政策类。字段名以 fixture 为准校准。"""
    cutoff = _window_start_ts(now)
    out = []
    for it in data.get("items", []) or data.get("data", []) or []:
        title = it.get("title") or ""
        ts = it.get("time") or it.get("ctime")
        if not any(k in title for k in POLICY_KEYWORDS):
            continue
        if ts and isinstance(ts, (int, float)) and ts < cutoff:
            continue
        out.append({"title": title.strip()[:80], "ts": ts or 0, "source": "新华", "star": 2})
    return out


def _similar(a, b):
    return SequenceMatcher(None, a, b).ratio() > 0.6


def merge_top(sources, topn=6):
    """多源合并 + 去重(标题相似度>0.6)+ 按重要性/时间排序,取 topn。"""
    pool = [it for src in sources for it in src]
    pool.sort(key=lambda x: (-x.get("star", 0), -x.get("ts", 0)))
    merged = []
    for it in pool:
        if any(_similar(it["title"], m["title"]) for m in merged):
            continue
        merged.append(it)
        if len(merged) >= topn:
            break
    return merged


def fetch_news():
    """返回 FetchResult(dim='news')。data={'items':[...], 'sources_ok':[...]}。"""
    now = datetime.now()
    sources_ok, all_items = [], []
    # 财联社

        cls_items = parse_cls(r.json(), now)
        all_items.append(cls_items)
        sources_ok.append("财联社")
    except Exception:
        all_items.append([])
    # 金十/新华:fixture 驱动,运行时同样尝试(反爬可能失败,失败则该源空)
    for parser, loader, name in [(parse_jin10, "jin10", "金十"), (parse_xinhua, "xinhua", "新华")]:
        try:
            # 运行时:金十/新华若无可靠 API,这里返回空(由 fixture test 保证 parser 正确)
            data = _safe_fetch(name)
            all_items.append(parser(data, now))
            if data:
                sources_ok.append(name)
        except Exception:
            all_items.append([])
    items = merge_top(all_items)
    ok = bool(items)
    detail = "✓" if ok else "⚠️ 新闻自动抓取失败(三源均无),请手动补充"
    return base.FetchResult("news", ok=ok, data={"items": items, "sources_ok": sources_ok},
                            detail=detail)
```

```python
静态人工整理的经验法则,非程序化回测。仅辅助研判,不据此下单。
tone() 给第0步"隔夜定调",signal_strength() 给信号强度。
阈值是经验值,可随沉淀调整。"""


def tone(us_pct, a50_pct, vix):
    """外部环境定调:偏暖/中性/偏冷/恐慌。基于美股+A50+VIX 综合。
    注意:这是'外部环境',不是'A股今天涨跌'。"""
    # 恐慌:VIX 飙升(>30) + 美股大跌(<-2)
    if vix and vix > 30 and us_pct < -2:
        return "恐慌"
    # 偏冷:美股跌>1.5 或 A50跌>0.5
    if us_pct < -1.5 or (a50_pct is not None and a50_pct < -0.5):
        return "偏冷"
    # 偏暖:美股涨>1
    if us_pct > 1:
        return "偏暖"
    return "中性"


def signal_strength(us_pct, a50_pct, vix):
    """信号强度:强/中/弱。多源同向=强。"""

    warm = us_pct > 1
    if (cold or warm) and vix and vix > 20:
        return "强"
    if cold or warm:
        return "中"
    return "弱"
```

```python
# -*- coding: utf-8 -*-
"""盘前外部温度计 (panqian) — 编排。
盘前采集 4 维隔夜信号 → 渲染「盘前外部温度计.md」。不预测涨跌。
对齐 fupan:屏蔽代理+trust_env=False、utf-8 stdout、argparse+步骤打印、写 output/{date}.md。

用法:
  python .claude/skills/panqian/main.py              # 默认盘前跑
  python .claude/skills/panqian/main.py --note "..."
"""
import argparse
import os
import sys
from datetime import datetime

from pk import config
from pk.base import FetchResult
from pk.us_stock import fetch_us
from pk.a50_cnr import fetch_a50_cnr
from pk.fx_commodity import fetch_fx_comm
from pk.news import fetch_news
from pk.mappings import tone, signal_strength
from pk.render import render

OUT_DIR = config.OUT_DIR

# Windows utf-8 输出(与 fupan 一致,防 emoji 崩)
try:
    sys.stdout.reconfigure(encoding="utf-8")
    sys.stderr.reconfigure(encoding="utf-8")
except Exception:
    pass

CRITICAL = ["us", "a50"]   # 关键维:缺失需报告顶部明示


def fetch_all():
    """采集 4 维,返回 {dim: FetchResult}。各维独立,互不阻断。"""
    res = {}
    print("[1/4] 美股+VIX ...")
    res["us"] = fetch_us()
    print(f"      {res['us'].detail}")
    print("[2/4] A50+中概 ...")
    res["a50"] = fetch_a50_cnr()
    print(f"      {res['a50'].detail}")
    print("[3/4] 汇率+大宗 ...")
    res["fx"] = fetch_fx_comm()
    print(f"      {res['fx'].detail}")
    print("[4/4] 国际新闻 ...")
    res["news"] = fetch_news()
    print(f"      {res['news'].detail}")
    return res


def _us_pct(res):
    idx = (res["us"].data or {}).get("indices", []) if res["us"].ok else []
    nas = next((i for i in idx if i["name"] == "纳指"), None)
    return nas["pct"] if nas else 0.0


def _a50_pct(res):
    if not res["a50"].ok:
        return None
    a50 = (res["a50"].data or {}).get("a50")
    return a50["pct"] if a50 else None


def build_report_data(date_str, res, note=""):
    """4 维 FetchResult → render 所需 dict。"""
    us_pct = _us_pct(res)
    a50_pct = _a50_pct(res)
    vix = (res["us"].data or {}).get("vix") if res["us"].ok else None
    critical_missing = any(not res[d].ok for d in CRITICAL)

    quality = {
        "us": res["us"].detail[:1] if res["us"].detail else ("✓" if res["us"].ok else "⚠️"),
        "a50": res["a50"].detail if not res["a50"].ok else "✓",
        "cnr": "✓" if (res["a50"].data or {}).get("cnr") else "-",
        "fx": "✓" if res["fx"].ok else "⚠️",
        "comm": "✓" if (res["fx"].data or {}).get("comm") else "-",
        "news": res["news"].detail if not res["news"].ok else "✓",
    }
    # 简化 quality 标记(用 ✓/⚠️)
    for k in ("us", "a50", "cnr", "fx", "comm", "news"):
        if "✓" in quality[k]:
            quality[k] = "✓"
        elif quality[k] in ("-", ""):
            quality[k] = "-"
        else:
            quality[k] = "⚠️"

    return {
        "date_str": date_str,
        "tone": tone(us_pct, a50_pct, vix or 0),
        "strength": signal_strength(us_pct, a50_pct, vix or 0),
        "us": res["us"].data if res["us"].ok else {"indices": [], "vix": None},
        "a50": res["a50"].data,
        "fx": res["fx"].data,
        "news": res["news"].data,
        "quality": quality,
        "critical_missing": critical_missing,
        "note": note,
    }


def run(date_str=None, note=""):
    date_str = date_str or datetime.now().strftime("%Y-%m-%d")
    print(f"[{date_str}] 盘前外部温度计\n")
    res = fetch_all()
    data = build_report_data(date_str, res, note)
    md = render(data)
    os.makedirs(OUT_DIR, exist_ok=True)
    out_path = os.path.join(OUT_DIR, f"{date_str}.md")
    with open(out_path, "w", encoding="utf-8") as f:
        f.write(md)
    print(f"\n报告 -> {out_path}")
    if data["critical_missing"]:
        print("⚠️ 关键维(A50/美股)缺失,报告已明示盲区")
    return out_path


def main():
    ap = argparse.ArgumentParser(description="盘前外部温度计")
    ap.add_argument("--note", default="", help="备注")
    args = ap.parse_args()
    run(note=args.note)


if __name__ == "__main__":
    main()
```

```python
# ---------- 6.5 隔夜信号对照 (读 panqian 盘前温度计) ----------
def _panqian_tone(pq_path):
    """从 panqian 报告抓'隔夜定调'一行,返回 (tone_str, a50_pct) 或 (None, None)。"""
    try:
        with open(pq_path, encoding="utf-8") as f:
            txt = f.read()
    except Exception:
        return None, None
    import re
    m = re.search(r"外部环境:\*\*(.+?)\*\*", txt)
    tone = m.group(1) if m else None
    a50m = re.search(r"A50期货:[\d.]+\(([+-]?[\d.]+)%\)", txt)
    a50_pct = float(a50m.group(1)) if a50m else None
    return tone, a50_pct


def realize_degree(predicted_low, open_pct):
    """兑现度:预示低开 vs 实际开盘。返回 兑现/背离/部分兑现。"""
    if predicted_low:
        if open_pct < -0.15:
            return "兑现"
        if open_pct > 0.15:
            return "背离"
        return "部分兑现"
    # 预示非低开(偏暖/中性),简化:看方向
    return "兑现" if open_pct > -0.15 else "背离"


def render_panqian_link(date_str, sh_open_pct, sh_close_pct):
    """渲染第-1步对照小节。读 panqian 当日报告,对照今日实际开盘/收盘。
    无 panqian 报告 → 降级提示。返回要插入剧本顶部的字符串。"""
    pq_path = os.path.join(PANQIAN_DIR, f"{date_str}.md")
    tone, a50_pct = _panqian_tone(pq_path)
    L = ["## -1. 隔夜信号 vs 今日实际(读自 panqian 盘前温度计)"]
    if tone is None:
        L.append("> ⚠️ 未找到今日 panqian 盘前温度计,建议盘前先跑 `/panqian`,以形成隔夜信号对照闭环。")
        return "\n".join(L) + "\n"
    predicted_low = a50_pct is not None and a50_pct < -0.3 or tone in ("偏冷", "恐慌")
    deg = realize_degree(predicted_low, sh_open_pct)
    L.append(f"- 隔夜预示:外部**{tone}**" +
             (f"(A50 {a50_pct:+.2f}% → 预示{'低开' if predicted_low else '平/高开'})" if a50_pct is not None else ""))
    L.append(f"- 今日实际:上证开盘 {sh_open_pct:+.2f}%｜收盘 {sh_close_pct:+.2f}%")
    L.append(f"- 兑现度:【{deg}】")
    L.append("- 教训沉淀:____(人工填,本次隔夜信号灵不灵)")
    return "\n".join(L) + "\n"
```
